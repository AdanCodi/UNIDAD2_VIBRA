package com.vibra.app

import android.Manifest
import android.content.Intent
import android.content.pm.PackageManager
import android.net.Uri
import android.os.Build
import android.os.Bundle
import androidx.activity.ComponentActivity
import androidx.activity.compose.rememberLauncherForActivityResult
import androidx.activity.compose.setContent
import androidx.activity.result.contract.ActivityResultContracts
import androidx.activity.viewModels
import androidx.compose.runtime.getValue
import androidx.compose.ui.platform.LocalContext
import androidx.core.content.ContextCompat
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import androidx.lifecycle.lifecycleScope
import com.vibra.app.ui.VibraApp
import com.vibra.app.wearlink.WatchLink
import kotlinx.coroutines.launch

class MainActivity : ComponentActivity() {
    private val viewModel: VibraViewModel by viewModels()
    private var activeWatchRequest: String? = null
    private var reportedPhase: ScanPhase? = null

    private val watchMicrophonePermission = registerForActivityResult(
        ActivityResultContracts.RequestPermission()
    ) { granted ->
        if (granted) beginWearRecognition()
        else {
            activeWatchRequest?.let {
                WatchLink.publish(this, it, "error", detail = "Permiso de micrófono rechazado")
            }
            activeWatchRequest = null
            viewModel.showError("El reloj necesita permiso para usar el micrófono del celular.")
        }
    }
    private val notificationPermission = registerForActivityResult(
        ActivityResultContracts.RequestPermission()
    ) { /* Si se deniega, el usuario puede abrir Vibra manualmente. */ }

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContent {
            val state by viewModel.state.collectAsStateWithLifecycle()
            val context = LocalContext.current
            val permission = rememberLauncherForActivityResult(
                ActivityResultContracts.RequestPermission()
            ) { granted ->
                if (granted) viewModel.startListening()
                else viewModel.showError("Debes permitir el micrófono para reconocer música.")
            }
            VibraApp(
                state = state,
                onListen = {
                    if (state.phase == ScanPhase.RECORDING) {
                        viewModel.cancelListening()
                    } else if (ContextCompat.checkSelfPermission(
                            context, Manifest.permission.RECORD_AUDIO
                        ) == PackageManager.PERMISSION_GRANTED
                    ) {
                        viewModel.startListening()
                    } else {
                        permission.launch(Manifest.permission.RECORD_AUDIO)
                    }
                },
                onSaveCredentials = viewModel::saveCredentials,
                onClearHistory = viewModel::clearHistory,
                onOpenLink = { link ->
                    val parsed = Uri.parse(link)
                    if (parsed.scheme == "https" || parsed.scheme == "http") {
                        runCatching { startActivity(Intent(Intent.ACTION_VIEW, parsed)) }
                    }
                },
                onShareSong = { song ->
                    val link = song.spotifyUrl ?: song.songLink
                    val details = "${song.title} — ${song.artist}" +
                        (if (link.isNullOrBlank()) "" else "\n$link")
                    startActivity(Intent.createChooser(Intent(Intent.ACTION_SEND).apply {
                        type = "text/plain"
                        putExtra(Intent.EXTRA_TEXT, details)
                    }, "Compartir canción"))
                }
            )
        }
        lifecycleScope.launch {
            viewModel.state.collect { state -> reportToWear(state) }
        }
        if (Build.VERSION.SDK_INT >= 33 && ContextCompat.checkSelfPermission(
                this, Manifest.permission.POST_NOTIFICATIONS
            ) != PackageManager.PERMISSION_GRANTED
        ) notificationPermission.launch(Manifest.permission.POST_NOTIFICATIONS)
    }

    override fun onResume() {
        super.onResume()
        handleIncomingWearRequest()
    }

    override fun onNewIntent(intent: Intent) {
        super.onNewIntent(intent)
        setIntent(intent)
        handleIncomingWearRequest()
    }

    private fun handleIncomingWearRequest() {
        val id = intent?.getStringExtra(WatchLink.EXTRA_REQUEST_ID)
            ?.takeIf { it.isNotBlank() }
            ?: WatchLink.takePending(this)
            ?: return
        intent?.removeExtra(WatchLink.EXTRA_REQUEST_ID)
        WatchLink.takePending(this) // Evita reactivar la misma solicitud.
        if (id == activeWatchRequest) return
        activeWatchRequest = id
        reportedPhase = null
        if (ContextCompat.checkSelfPermission(this, Manifest.permission.RECORD_AUDIO) ==
            PackageManager.PERMISSION_GRANTED
        ) beginWearRecognition()
        else watchMicrophonePermission.launch(Manifest.permission.RECORD_AUDIO)
    }

    private fun beginWearRecognition() {
        val id = activeWatchRequest ?: return
        WatchLink.publish(this, id, "recording", detail = "Escuchando desde el celular")
        viewModel.startListening()
    }

    private fun reportToWear(state: VibraState) {
        val id = activeWatchRequest ?: return
        if (reportedPhase == state.phase) return
        when (state.phase) {
            ScanPhase.RECORDING -> WatchLink.publish(this, id, "recording", detail = "Grabando 11 segundos")
            ScanPhase.IDENTIFYING -> WatchLink.publish(this, id, "identifying", detail = "Buscando canción")
            ScanPhase.FOUND -> {
                val song = state.song
                WatchLink.publish(this, id, "found", song?.title.orEmpty(), song?.artist.orEmpty())
                activeWatchRequest = null
            }
            ScanPhase.NO_MATCH -> {
                WatchLink.publish(this, id, "no_match", detail = "Sin coincidencias")
                activeWatchRequest = null
            }
            ScanPhase.ERROR -> {
                WatchLink.publish(this, id, "error", detail = state.errorMessage.orEmpty())
                activeWatchRequest = null
            }
            ScanPhase.IDLE -> Unit
        }
        reportedPhase = state.phase
    }

    override fun onStop() {
        if (activeWatchRequest != null && viewModel.state.value.phase == ScanPhase.RECORDING) {
            WatchLink.publish(this, activeWatchRequest!!, "error", detail = "Grabación interrumpida")
            activeWatchRequest = null
        }
        viewModel.cancelListening() // No mantener el micrófono abierto en segundo plano.
        super.onStop()
    }
}

package com.vibra.app

import android.app.Application
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import com.vibra.app.audio.MicrophoneRecorder
import com.vibra.app.data.AcrCloudClient
import com.vibra.app.data.AcrCredentials
import com.vibra.app.data.Song
import com.vibra.app.data.VibraPreferences
import kotlinx.coroutines.CancellationException
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.Job
import kotlinx.coroutines.delay
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.launch
import kotlinx.coroutines.withContext

private const val RECORD_SECONDS = 11

enum class ScanPhase { IDLE, RECORDING, IDENTIFYING, FOUND, NO_MATCH, ERROR }

data class VibraState(
    val phase: ScanPhase = ScanPhase.IDLE,
    val secondsRemaining: Int = RECORD_SECONDS,
    val inputLevel: Float = 0f,
    val song: Song? = null,
    val errorMessage: String? = null,
    val history: List<Song> = emptyList(),
    val acrCredentials: AcrCredentials = AcrCredentials()
)

class VibraViewModel(application: Application) : AndroidViewModel(application) {
    private val preferences = VibraPreferences(application)
    private val recorder = MicrophoneRecorder(application)
    private val client = AcrCloudClient()
    private val mutableState = MutableStateFlow(
        VibraState(history = preferences.getHistory(), acrCredentials = preferences.acrCredentials)
    )
    val state = mutableState.asStateFlow()
    private var recordJob: Job? = null

    fun startListening() {
        if (mutableState.value.phase == ScanPhase.RECORDING ||
            mutableState.value.phase == ScanPhase.IDENTIFYING) return
        val credentials = preferences.acrCredentials
        if (!credentials.isConfigured()) {
            showError("Primero agrega tu Host, Access Key y Access Secret de ACRCloud en Ajustes.")
            return
        }
        recordJob = viewModelScope.launch {
            var clip: java.io.File? = null
            try {
                recorder.start()
                mutableState.value = mutableState.value.copy(
                    phase = ScanPhase.RECORDING,
                    secondsRemaining = RECORD_SECONDS,
                    inputLevel = 0f,
                    song = null,
                    errorMessage = null
                )
                // Actualizamos el medidor 5 veces por segundo para comprobar si el
                // hardware realmente está entregando sonido durante la grabación.
                repeat(RECORD_SECONDS * 5) { tick ->
                    delay(200L)
                    mutableState.value = mutableState.value.copy(
                        secondsRemaining = (RECORD_SECONDS - (tick + 1) / 5).coerceAtLeast(0),
                        inputLevel = recorder.inputLevel
                    )
                }
                val recordedFile = recorder.stop()
                clip = recordedFile
                mutableState.value = mutableState.value.copy(phase = ScanPhase.IDENTIFYING)
                val song = withContext(Dispatchers.IO) { client.identify(recordedFile, credentials) }
                if (song == null) {
                    mutableState.value = mutableState.value.copy(phase = ScanPhase.NO_MATCH)
                } else {
                    preferences.saveSong(song)
                    mutableState.value = mutableState.value.copy(
                        phase = ScanPhase.FOUND, song = song, history = preferences.getHistory()
                    )
                }
            } catch (cancelled: CancellationException) {
                throw cancelled
            } catch (error: SecurityException) {
                showError("Sin permiso para usar el micrófono.")
            } catch (error: Exception) {
                showError(error.message ?: "No fue posible reconocer la canción.")
            } finally {
                recorder.discard()
                clip?.delete()
            }
        }
    }

    fun cancelListening() {
        if (mutableState.value.phase == ScanPhase.RECORDING) {
            recordJob?.cancel()
            recorder.discard()
            mutableState.value = mutableState.value.copy(phase = ScanPhase.IDLE)
        }
    }

    fun showError(message: String) {
        mutableState.value = mutableState.value.copy(phase = ScanPhase.ERROR, errorMessage = message)
    }

    fun saveCredentials(credentials: AcrCredentials) {
        preferences.acrCredentials = credentials
        mutableState.value = mutableState.value.copy(acrCredentials = preferences.acrCredentials)
    }

    fun clearHistory() {
        preferences.clearHistory()
        mutableState.value = mutableState.value.copy(history = emptyList())
    }

    override fun onCleared() {
        recordJob?.cancel()
        recorder.discard()
        super.onCleared()
    }
}

package com.vibra.app.audio

import android.annotation.SuppressLint
import android.content.Context
import android.media.AudioFormat
import android.media.AudioRecord
import android.media.MediaRecorder
import android.os.Build
import java.io.File
import java.io.FileOutputStream
import java.io.IOException
import java.io.RandomAccessFile
import kotlin.math.sqrt
import kotlin.math.log10

/**
 * Graba el micrófono en WAV PCM mono, sin códec AAC.
 * Un hilo propio vacía el búfer de AudioRecord mientras la UI sigue respondiendo.
 * El ViewModel conserva el control de la duración de la captura (11 segundos).
 */
class MicrophoneRecorder(private val context: Context) {
    private companion object {
        const val SAMPLE_RATE = 44_100
        const val WAV_HEADER_BYTES = 44L
        const val MIN_SECONDS = 6
        // Solo bloqueamos capturas prácticamente digitales en cero. Un umbral fijo alto
        // puede rechazar música real cuando el hardware usa una ganancia de micrófono baja.
        const val DIGITAL_SILENCE_RMS = 2.0
    }

    @Volatile private var isCapturing = false
    @Volatile private var workerError: Throwable? = null
    private var recorder: AudioRecord? = null
    private var worker: Thread? = null
    private var outputFile: File? = null
    private var capturedBytes = 0L
    private var sumOfSquares = 0.0
    private var sampleCount = 0L
    private var maxSample = 0

    // Medidor de entrada en vivo (0-1), publicado sin acceder al archivo de audio.
    @Volatile var inputLevel: Float = 0f
        private set

    @SuppressLint("MissingPermission") // El permiso se solicita antes desde MainActivity.
    @Synchronized
    fun start() {
        check(recorder == null) { "Ya hay una grabación activa" }
        val minBuffer = AudioRecord.getMinBufferSize(
            SAMPLE_RATE,
            AudioFormat.CHANNEL_IN_MONO,
            AudioFormat.ENCODING_PCM_16BIT
        )
        check(minBuffer > 0) { "El dispositivo no admite el formato de micrófono requerido" }

        val input = AudioRecord(
            MediaRecorder.AudioSource.MIC,
            SAMPLE_RATE,
            AudioFormat.CHANNEL_IN_MONO,
            AudioFormat.ENCODING_PCM_16BIT,
            maxOf(minBuffer * 2, 16_384)
        )
        if (input.state != AudioRecord.STATE_INITIALIZED) {
            input.release()
            throw IllegalStateException("No se pudo inicializar el micrófono")
        }
        val file = File.createTempFile("vibra_", ".wav", context.cacheDir)
        try {
            input.startRecording()
            check(input.recordingState == AudioRecord.RECORDSTATE_RECORDING) {
                "Android no permitió comenzar a grabar"
            }
            capturedBytes = 0L
            sampleCount = 0L
            sumOfSquares = 0.0
            maxSample = 0
            inputLevel = 0f
            workerError = null
            outputFile = file
            recorder = input
            isCapturing = true

            val thread = Thread({
                try {
                    FileOutputStream(file).buffered().use { out ->
                        out.write(ByteArray(WAV_HEADER_BYTES.toInt()))
                        val buffer = ByteArray(8_192)
                        while (isCapturing) {
                            val bytes = input.read(buffer, 0, buffer.size)
                            when {
                                bytes > 0 -> {
                                    val evenBytes = bytes - bytes % 2
                                    out.write(buffer, 0, evenBytes)
                                    capturedBytes += evenBytes
                                    var offset = 0
                                    var blockSumSquares = 0.0
                                    var blockCount = 0
                                    while (offset < evenBytes) {
                                        val sample = (
                                            ((buffer[offset + 1].toInt() shl 8) or
                                                (buffer[offset].toInt() and 0xff))
                                            ).toShort().toInt()
                                        val square = sample.toDouble() * sample
                                        sumOfSquares += square
                                        blockSumSquares += square
                                        sampleCount++
                                        blockCount++
                                        val amplitude = kotlin.math.abs(sample)
                                        if (amplitude > maxSample) maxSample = amplitude
                                        offset += 2
                                    }
                                    if (blockCount > 0) {
                                        val blockRms = sqrt(blockSumSquares / blockCount)
                                        val dBfs = 20.0 * log10((blockRms / 32768.0).coerceAtLeast(1e-9))
                                        inputLevel = ((dBfs + 65.0) / 65.0).coerceIn(0.0, 1.0).toFloat()
                                    }
                                }
                                !isCapturing -> break
                                bytes == 0 -> Thread.yield()
                                else -> throw IOException("El micrófono devolvió el código $bytes")
                            }
                        }
                    }
                } catch (error: Throwable) {
                    workerError = error
                } finally {
                    isCapturing = false
                }
            }, "VibraAudioCapture")
            worker = thread
            thread.start()
        } catch (error: Exception) {
            isCapturing = false
            recorder = null
            outputFile = null
            runCatching { input.stop() }
            input.release()
            file.delete()
            throw error
        }
    }

    @Synchronized
    fun stop(): File {
        val file = outputFile ?: throw IllegalStateException("No hay grabación activa")
        // Android 10+ puede entregar ceros cuando otra captura tiene prioridad.
        val clientSilenced = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
            runCatching { recorder?.activeRecordingConfiguration?.isClientSilenced == true }
                .getOrDefault(false)
        } else false
        stopHardware()
        workerError?.let {
            file.delete()
            throw IOException("Falló la captura del micrófono: ${it.message}", it)
        }
        val bytes = capturedBytes
        val duration = bytes / (SAMPLE_RATE * 2L)
        if (duration < MIN_SECONDS) {
            file.delete()
            throw IllegalStateException("Se grabaron solo $duration segundos. Prueba de nuevo")
        }
        val rms = if (sampleCount > 0) sqrt(sumOfSquares / sampleCount) else 0.0
        if (clientSilenced) {
            file.delete()
            throw IllegalStateException(
                "Android silenció el micrófono de Vibra. Cierra otras apps que estén grabando " +
                    "y deja Vibra abierta en primer plano durante la escucha."
            )
        }
        if (rms < DIGITAL_SILENCE_RMS || maxSample <= 2) {
            file.delete()
            throw IllegalStateException(
                "La grabación está prácticamente vacía (nivel RMS=${rms.toInt()}, " +
                    "pico=$maxSample). Comprueba que el acceso global al micrófono esté " +
                    "activado y prueba a grabar una nota de voz. Si utilizas un emulador, " +
                    "prueba Vibra en un teléfono real."
            )
        }
        // El audio tenue pero real se envía a ACRCloud. Evitamos rechazarlo con el antiguo MIN_RMS=100.
        if (bytes > Int.MAX_VALUE - WAV_HEADER_BYTES) {
            file.delete()
            throw IOException("El archivo de audio excede el tamaño permitido")
        }
        writeWavHeader(file, bytes.toInt())
        return file
    }

    @Synchronized
    fun discard() {
        stopHardware()
        outputFile?.delete()
        outputFile = null
    }

    private fun stopHardware() {
        isCapturing = false
        val input = recorder
        recorder = null
        if (input != null) {
            runCatching { input.stop() }
        }
        val thread = worker
        if (thread != null) {
            try {
                thread.join(3_000)
            } catch (ex: InterruptedException) {
                Thread.currentThread().interrupt()
                workerError = ex
            }
        }
        worker = null
        input?.release()
        if (thread?.isAlive == true) {
            throw IOException("La grabación no pudo finalizar correctamente")
        }
    }

    private fun writeWavHeader(file: File, pcmBytes: Int) {
        val byteRate = SAMPLE_RATE * 2 // mono, 16 bits
        RandomAccessFile(file, "rw").use { wav ->
            wav.seek(0)
            fun ascii(value: String) = wav.write(value.toByteArray(Charsets.US_ASCII))
            fun write16LE(value: Int) {
                wav.write(value and 0xff)
                wav.write((value ushr 8) and 0xff)
            }
            fun write32LE(value: Int) {
                for (shift in 0..24 step 8) wav.write((value ushr shift) and 0xff)
            }
            ascii("RIFF")
            write32LE(36 + pcmBytes)
            ascii("WAVE")
            ascii("fmt ")
            write32LE(16)
            write16LE(1) // PCM
            write16LE(1) // 1 canal
            write32LE(SAMPLE_RATE)
            write32LE(byteRate)
            write16LE(2) // block align
            write16LE(16) // bits por muestra
            ascii("data")
            write32LE(pcmBytes)
        }
    }
}

package com.vibra.app.data

import org.json.JSONArray
import org.json.JSONObject
import java.io.ByteArrayOutputStream
import java.io.File
import java.io.IOException
import java.net.HttpURLConnection
import java.net.URL
import java.util.UUID
import javax.crypto.Mac
import javax.crypto.spec.SecretKeySpec
import java.util.Base64

/** Configuración del proyecto de reconocimiento musical de ACRCloud. */
data class AcrCredentials(
    val host: String = "",
    val accessKey: String = "",
    val accessSecret: String = ""
) {
    fun isConfigured(): Boolean = host.isNotBlank() && accessKey.isNotBlank() && accessSecret.isNotBlank()

    fun normalizedHost(): String = host.trim()
        .removePrefix("https://")
        .trimEnd('/')
        .removeSuffix("/v1/identify")
        .lowercase()
}

/** API REST de ACRCloud v1: WAV 11 s + HMAC-SHA1 multipart HTTPS. */
class AcrCloudClient {
    fun identify(recording: File, credentials: AcrCredentials): Song? {
        require(credentials.isConfigured()) { "Configura Host, Access Key y Access Secret de ACRCloud en Ajustes." }
        val host = credentials.normalizedHost()
        if (!Regex("^[a-z0-9-]+(?:\\.[a-z0-9-]+)*\\.acrcloud\\.com$").matches(host)) {
            throw IllegalArgumentException("El Host debe ser el del proyecto ACRCloud, por ejemplo identify-eu-west-1.acrcloud.com")
        }
        if (!recording.isFile || recording.length() <= 44L) {
            throw IOException("La grabación está vacía")
        }
        if (recording.length() >= 5_000_000L) {
            throw IOException("La muestra supera el máximo de 5 MB de ACRCloud")
        }
        val header = ByteArray(12)
        recording.inputStream().use { input ->
            if (input.read(header) != header.size ||
                String(header, 0, 4, Charsets.US_ASCII) != "RIFF" ||
                String(header, 8, 4, Charsets.US_ASCII) != "WAVE") {
                throw IOException("El micrófono no produjo un WAV válido")
            }
        }

        val timestamp = (System.currentTimeMillis() / 1000L).toString()
        val accessKey = credentials.accessKey.trim()
        val dataToSign = "POST\n/v1/identify\n$accessKey\naudio\n1\n$timestamp"
        val signer = Mac.getInstance("HmacSHA1")
        signer.init(SecretKeySpec(credentials.accessSecret.trim().toByteArray(Charsets.UTF_8), "HmacSHA1"))
        val signature = Base64.getEncoder().encodeToString(signer.doFinal(dataToSign.toByteArray(Charsets.UTF_8)))
        val boundary = "Vibra${UUID.randomUUID()}"
        val prefix = ByteArrayOutputStream().apply {
            fun put(value: String) = write(value.toByteArray(Charsets.UTF_8))
            fun field(name: String, value: String) {
                put("--$boundary\r\n")
                put("Content-Disposition: form-data; name=\"$name\"\r\n\r\n")
                put(value)
                put("\r\n")
            }
            field("access_key", accessKey)
            field("sample_bytes", recording.length().toString())
            field("timestamp", timestamp)
            field("signature", signature)
            field("data_type", "audio")
            field("signature_version", "1")
            put("--$boundary\r\n")
            put("Content-Disposition: form-data; name=\"sample\"; filename=\"vibra.wav\"\r\n")
            put("Content-Type: audio/wav\r\n\r\n")
        }.toByteArray()
        val suffix = "\r\n--$boundary--\r\n".toByteArray(Charsets.UTF_8)
        val connection = URL("https://$host/v1/identify").openConnection() as HttpURLConnection
        try {
            connection.requestMethod = "POST"
            connection.connectTimeout = 20_000
            connection.readTimeout = 40_000
            connection.doOutput = true
            connection.setRequestProperty("Content-Type", "multipart/form-data; boundary=$boundary")
            connection.setRequestProperty("Accept", "application/json")
            connection.setFixedLengthStreamingMode(prefix.size.toLong() + recording.length() + suffix.size)
            connection.outputStream.use { out ->
                out.write(prefix)
                recording.inputStream().buffered().use { input -> input.copyTo(out) }
                out.write(suffix)
            }
            val httpStatus = connection.responseCode
            val stream = if (httpStatus in 200..299) connection.inputStream else connection.errorStream
            val raw = stream?.bufferedReader()?.use { it.readText() }
                ?: throw IOException("ACRCloud no respondió con datos (HTTP $httpStatus)")
            val root = try { JSONObject(raw) } catch (ex: Exception) {
                throw IOException("ACRCloud devolvió una respuesta inválida (HTTP $httpStatus)", ex)
            }
            val status = root.optJSONObject("status")
            val code = status?.optInt("code", -1) ?: -1
            if (code == 1001) return null // No Result: respuesta normal cuando no hay coincidencia.
            if (httpStatus !in 200..299 || code != 0) {
                val reason = status?.optString("msg").orEmpty().take(160)
                val explanation = when (code) {
                    3001, 3014 -> "Credenciales incorrectas o servicio no autorizado"
                    2004, 2005, 2006 -> "ACRCloud no pudo analizar el audio; comprueba el micrófono"
                    else -> reason.ifBlank { "Revisa las credenciales, el Host y la cuota de ACRCloud" }
                }
                throw IOException("ACRCloud: $explanation (código $code, HTTP $httpStatus)")
            }
            val music = root.optJSONObject("metadata")?.optJSONArray("music") ?: return null
            val match = music.optJSONObject(0) ?: return null
            val title = match.optString("title").ifBlank { "Canción desconocida" }
            val artists = match.optJSONArray("artists")
            val artist = (0 until (artists?.length() ?: 0))
                .mapNotNull { artists?.optJSONObject(it)?.optNullableString("name") }
                .joinToString(", ").ifBlank { "Artista desconocido" }
            val album = match.optJSONObject("album")?.optString("name").orEmpty()
            val metadata = match.optJSONObject("external_metadata")
            val spotify = firstJson(metadata?.opt("spotify"))
            val spotifyId = spotify?.optJSONObject("track")?.optNullableString("id")
            val spotifyUrl = if (spotifyId != null && Regex("^[A-Za-z0-9]{22}$").matches(spotifyId)) {
                "https://open.spotify.com/track/$spotifyId"
            } else null
            // ACRCloud no garantiza portada en el resultado básico: la UI conserva su icono de respaldo.
            return Song(
                title = title,
                artist = artist,
                album = album,
                artworkUrl = null,
                songLink = spotifyUrl,
                spotifyUrl = spotifyUrl
            )
        } finally {
            connection.disconnect()
        }
    }

    private fun firstJson(value: Any?): JSONObject? = when (value) {
        is JSONObject -> value
        is JSONArray -> value.optJSONObject(0)
        else -> null
    }
}

package com.vibra.app.data

import org.json.JSONObject

data class Song(
    val title: String,
    val artist: String,
    val album: String,
    val artworkUrl: String?,
    val songLink: String?,
    val spotifyUrl: String?,
    val recognizedAt: Long = System.currentTimeMillis()
) {
    fun toJson(): JSONObject = JSONObject().apply {
        put("title", title)
        put("artist", artist)
        put("album", album)
        put("artworkUrl", artworkUrl)
        put("songLink", songLink)
        put("spotifyUrl", spotifyUrl)
        put("recognizedAt", recognizedAt)
    }

    companion object {
        fun fromJson(json: JSONObject): Song = Song(
            title = json.optString("title", "Desconocida"),
            artist = json.optString("artist", "Artista desconocido"),
            album = json.optString("album", "Sin álbum"),
            artworkUrl = json.optNullableString("artworkUrl"),
            songLink = json.optNullableString("songLink"),
            spotifyUrl = json.optNullableString("spotifyUrl"),
            recognizedAt = json.optLong("recognizedAt", System.currentTimeMillis())
        )
    }
}

internal fun JSONObject.optNullableString(key: String): String? =
    if (isNull(key)) null else optString(key).takeIf { it.isNotBlank() && it != "null" }

package com.vibra.app.data

import android.content.Context
import org.json.JSONArray

/** Demostración local. Para producción, el access_secret debe protegerse en un backend. */
class VibraPreferences(context: Context) {
    private val prefs = context.applicationContext.getSharedPreferences("vibra", Context.MODE_PRIVATE)

    var acrCredentials: AcrCredentials
        get() = AcrCredentials(
            host = prefs.getString("acr_host", "").orEmpty(),
            accessKey = prefs.getString("acr_key", "").orEmpty(),
            accessSecret = prefs.getString("acr_secret", "").orEmpty()
        )
        set(value) {
            prefs.edit()
                .putString("acr_host", value.normalizedHost())
                .putString("acr_key", value.accessKey.trim())
                .putString("acr_secret", value.accessSecret.trim())
                .apply()
        }

    fun getHistory(): List<Song> = runCatching {
        val array = JSONArray(prefs.getString("history", "[]"))
        (0 until array.length()).mapNotNull { index ->
            runCatching { Song.fromJson(array.getJSONObject(index)) }.getOrNull()
        }.sortedByDescending { it.recognizedAt }
    }.getOrDefault(emptyList())

    fun saveSong(song: Song) {
        val current = getHistory().filterNot {
            it.title.equals(song.title, ignoreCase = true) &&
                it.artist.equals(song.artist, ignoreCase = true)
        }
        val array = JSONArray()
        (listOf(song) + current).take(50).forEach { array.put(it.toJson()) }
        prefs.edit().putString("history", array.toString()).apply()
    }

    fun clearHistory() {
        prefs.edit().remove("history").apply()
    }
}

private enum class Page { HOME, HISTORY, SETTINGS, RESULT }

@Composable
fun VibraApp(
    state: VibraState,
    onListen: () -> Unit,
    onSaveCredentials: (AcrCredentials) -> Unit,
    onClearHistory: () -> Unit,
    onOpenLink: (String) -> Unit,
    onShareSong: (Song) -> Unit = {}
) {
    var page by rememberSaveable { mutableStateOf(Page.HOME) }
    var resultSource by rememberSaveable { mutableStateOf(Page.HOME) }
    var shownSong by remember { mutableStateOf<Song?>(null) }
    var lastShownRecognition by rememberSaveable { mutableStateOf<Long?>(null) }

    // Cuando se reconoce una canción se abre la pantalla de resultados automáticamente.
    LaunchedEffect(state.phase, state.song?.recognizedAt) {
        val song = state.song
        if (state.phase == ScanPhase.FOUND && song != null && lastShownRecognition != song.recognizedAt) {
            shownSong = song
            lastShownRecognition = song.recognizedAt
            resultSource = Page.HOME
            page = Page.RESULT
        }
    }

    BackHandler(enabled = page == Page.RESULT) { page = resultSource }
    MaterialTheme(colorScheme = VibraColors) {
        Box(Modifier.fillMaxSize()) {
            AtmosphericBackground()
            Scaffold(
                containerColor = Color.Transparent,
                bottomBar = {
                    if (page != Page.RESULT) {
                        VibraNavigationBar(page) { page = it }
                    }
                }
            ) { scaffoldPadding ->
                Box(
                    modifier = Modifier.fillMaxSize().padding(scaffoldPadding),
                    contentAlignment = Alignment.TopCenter
                ) {
                    Column(
                        Modifier.fillMaxWidth().widthIn(max = 560.dp)
                            .padding(horizontal = 22.dp)
                    ) {
                        when (page) {
                            Page.HOME -> HomePage(
                                state = state,
                                onListen = onListen,
                                onSettings = { page = Page.SETTINGS }
                            )
                            Page.HISTORY -> HistoryPage(
                                history = state.history,
                                onClear = onClearHistory,
                                onSelect = { song ->
                                    shownSong = song
                                    resultSource = Page.HISTORY
                                    page = Page.RESULT
                                }
                            )
                            Page.SETTINGS -> SettingsPage(state.acrCredentials, onSaveCredentials)
                            Page.RESULT -> shownSong?.let { song ->
                                ResultPage(
                                    song = song,
                                    onBack = { page = resultSource },
                                    onOpenLink = onOpenLink,
                                    onShare = { onShareSong(song) }
                                )
                            }
                        }
                    }
                }
            }
        }
    }
}

            }
            drawCircle(
                brush = Brush.linearGradient(listOf(Cyan, Electric, Violet),

@Composable
private fun AudioVisualizer(active: Boolean, level: Float) {
    val movement = rememberInfiniteTransition(label = "ecualizador")
    val oscillation by movement.animateFloat(
        0f, 1f, infiniteRepeatable(tween(600), RepeatMode.Reverse), label = "barras"
    )
    Row(
        Modifier.fillMaxWidth().height(42.dp),
        horizontalArrangement = Arrangement.Center, verticalAlignment = Alignment.CenterVertically
    ) {
        repeat(27) { i ->
            val normalized = abs(i - 13) / 13f
            val shape = (1f - normalized * .87f).coerceAtLeast(.08f)
            val activity = if (active) .35f + .65f * level.coerceIn(0f, 1f) else .62f
            val variation = if (active) .7f + .3f * abs(sin((i + 1) * .7f + oscillation * 6f)) else 1f
            val barHeight = (5f + 33f * shape * activity * variation).dp
            Box(
                Modifier.padding(horizontal = 1.6.dp).width(3.1.dp).height(barHeight)
                    .clip(CircleShape)
                    .background(if (i < 13) Cyan else if (i < 18) Electric else Violet)
            )
        }
    }

package com.vibra.app.wearlink

import com.google.android.gms.wearable.MessageEvent
import com.google.android.gms.wearable.WearableListenerService
import org.json.JSONObject

/** Recibe una solicitud únicamente desde la aplicación vinculada por Data Layer. */
class PhoneWatchListenerService : WearableListenerService() {
    override fun onMessageReceived(event: MessageEvent) {
        if (event.path != WatchLink.REQUEST_PATH || event.data.size > 1024) return
        val id = runCatching {
            JSONObject(String(event.data, Charsets.UTF_8)).optString("id")
        }.getOrDefault("")
        if (!Regex("^[a-zA-Z0-9-]{8,64}$").matches(id)) return
        WatchLink.savePending(this, id)
        WatchLink.publish(this, id, "waiting", detail = "Abre la solicitud en tu celular")
        WatchLink.showRequestNotification(this, id)
    }
}

package com.vibra.app.wearlink

import android.Manifest
import android.app.NotificationChannel
import android.app.NotificationManager
import android.app.PendingIntent
import android.content.Context
import android.content.Intent
import android.content.pm.PackageManager
import android.os.Build
import androidx.core.app.NotificationCompat
import androidx.core.content.ContextCompat
import com.google.android.gms.wearable.PutDataMapRequest
import com.google.android.gms.wearable.Wearable
import com.vibra.app.MainActivity
import com.vibra.app.R

/**
 * Comunicación reloj-celular. Solo se envían estados y metadatos musicales;
 * nunca audio, credenciales de ACRCloud ni identificadores personales.
 */
object WatchLink {
    const val REQUEST_PATH = "/vibra/request"
    const val STATE_PATH = "/vibra/state"
    const val EXTRA_REQUEST_ID = "vibra.request.id"
    private const val PREFS = "vibra_wear_messages"
    private const val PENDING = "pending_request_id"
    private const val CHANNEL = "vibra_wear_requests"

    fun savePending(context: Context, id: String) {
        context.getSharedPreferences(PREFS, Context.MODE_PRIVATE)
            .edit().putString(PENDING, id).apply()
    }

    fun takePending(context: Context): String? {
        val p = context.getSharedPreferences(PREFS, Context.MODE_PRIVATE)
        val id = p.getString(PENDING, null)
        if (id != null) p.edit().remove(PENDING).apply()
        return id
    }

    fun publish(
        context: Context, id: String, stage: String,
        title: String = "", artist: String = "", detail: String = ""
    ) {
        val data = PutDataMapRequest.create(STATE_PATH).apply {
            dataMap.putString("id", id)
            dataMap.putString("stage", stage)
            dataMap.putString("title", title.take(140))
            dataMap.putString("artist", artist.take(140))
            dataMap.putString("detail", detail.take(180))
            dataMap.putLong("updatedAt", System.currentTimeMillis())
        }.asPutDataRequest().setUrgent()
        Wearable.getDataClient(context.applicationContext).putDataItem(data)
    }

    /** No se intenta grabar audio desde un servicio en segundo plano. */
    fun showRequestNotification(context: Context, id: String) {
        val manager = context.getSystemService(NotificationManager::class.java)
        manager.createNotificationChannel(
            NotificationChannel(CHANNEL, "Solicitudes desde reloj", NotificationManager.IMPORTANCE_HIGH)
        )
        val open = Intent(context, MainActivity::class.java).apply {
            flags = Intent.FLAG_ACTIVITY_SINGLE_TOP or Intent.FLAG_ACTIVITY_CLEAR_TOP
            putExtra(EXTRA_REQUEST_ID, id)
        }
        val pending = PendingIntent.getActivity(
            context, 19, open, PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )
        val allowed = Build.VERSION.SDK_INT < 33 ||
            ContextCompat.checkSelfPermission(context, Manifest.permission.POST_NOTIFICATIONS) == PackageManager.PERMISSION_GRANTED
        if (!allowed) return
        manager.notify(2019, NotificationCompat.Builder(context, CHANNEL)
            .setSmallIcon(R.mipmap.ic_launcher)
            .setContentTitle("Vibra Watch quiere reconocer música")
            .setContentText("Toca para abrir Vibra y activar el micrófono del celular")
            .setContentIntent(pending)
            .setAutoCancel(true)
            .setPriority(NotificationCompat.PRIORITY_HIGH)
            .build())
    }
}

private const val REQUEST = "/vibra/request"
private const val STATE = "/vibra/state"
private val Cyan = Color(0xFF26D9FF)
private val Violet = Color(0xFF9E52FF)
private val Ice = Color(0xFFF7F9FF)
private val Muted = Color(0xFFADB9DC)

private data class WatchState(
    val connected: Boolean = false,
    val stage: String = "idle",
    val title: String = "",
    val artist: String = "",
    val detail: String = "Toca para reconocer música",
    val requestId: String = ""
)

class MainActivity : ComponentActivity(), DataClient.OnDataChangedListener {
    private var state by mutableStateOf(WatchState())

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContent { WatchScreen(state = state, onRecognize = ::recognize) }
    }

    override fun onResume() {
        super.onResume()
        Wearable.getDataClient(this).addListener(this)
        Wearable.getNodeClient(this).connectedNodes
            .addOnSuccessListener { nodes ->
                state = state.copy(connected = nodes.isNotEmpty())
            }
            .addOnFailureListener {
                state = state.copy(connected = false, detail = "Google Play Services no disponible")
            }
        Wearable.getDataClient(this).dataItems.addOnSuccessListener { buffer ->
            try {
                for (item in buffer) if (item.uri.path == STATE) readUpdate(item)
            } finally { buffer.release() }
        }
    }

    override fun onPause() {
        Wearable.getDataClient(this).removeListener(this)
        super.onPause()
    }

    override fun onDataChanged(events: DataEventBuffer) {
        try {
            for (event in events) {
                if (event.type == com.google.android.gms.wearable.DataEvent.TYPE_CHANGED &&
                    event.dataItem.uri.path == STATE
                ) readUpdate(event.dataItem)
            }
        } finally { events.release() }
    }

    private fun readUpdate(item: com.google.android.gms.wearable.DataItem) {
        val data = DataMapItem.fromDataItem(item).dataMap
        val id = data.getString("id").orEmpty()
        // Si hay una solicitud en curso, ignorar respuestas atrasadas de otra solicitud.
        if (state.requestId.isNotBlank() && state.requestId != id) return
        state = state.copy(
            stage = data.getString("stage").orEmpty(),
            title = data.getString("title").orEmpty(),
            artist = data.getString("artist").orEmpty(),
            detail = data.getString("detail").orEmpty(),
            requestId = id
        )
    }

    private fun recognize() {
        if (state.stage in listOf("requesting", "waiting", "recording", "identifying")) return
        state = state.copy(stage = "requesting", detail = "Buscando teléfono...")
        Wearable.getNodeClient(this).connectedNodes.addOnSuccessListener { nodes ->
            val phone = nodes.firstOrNull { it.isNearby } ?: nodes.firstOrNull()
            if (phone == null) {
                state = state.copy(connected = false, stage = "error", detail = "Vincula un celular Android")
                return@addOnSuccessListener
            }
            val id = UUID.randomUUID().toString()
            state = state.copy(connected = true, requestId = id, stage = "requesting", title = "", artist = "", detail = "Enviando solicitud...")
            val payload = JSONObject().put("id", id).toString().toByteArray(Charsets.UTF_8)
            Wearable.getMessageClient(this).sendMessage(phone.id, REQUEST, payload)
                .addOnSuccessListener {
                    state = state.copy(stage = "waiting", detail = "Abre la notificación en el celular")
                }
                .addOnFailureListener {
                    state = state.copy(stage = "error", detail = "No se pudo contactar al celular")
                }
        }.addOnFailureListener {
            state = state.copy(connected = false, stage = "error", detail = "Sin conexión con el celular")
        }
    }
}

@Composable
private fun WatchScreen(state: WatchState, onRecognize: () -> Unit) {
    val listening = state.stage == "recording" || state.stage == "identifying"
    val transition = rememberInfiniteTransition(label = "pulse")
    val pulse by transition.animateFloat(
        initialValue = 0.82f, targetValue = 1.0f,
        animationSpec = infiniteRepeatable(tween(1300), RepeatMode.Reverse), label = "pulseFloat"
    )
    Box(
        Modifier.fillMaxSize().background(
            Brush.radialGradient(listOf(Color(0xFF10276A), Color(0xFF080D2B), Color(0xFF040715)))
        ), contentAlignment = Alignment.Center
    ) {
        Canvas(Modifier.fillMaxSize()) {
            drawCircle(color = Cyan.copy(alpha = if (listening) .23f * pulse else .07f),
                radius = size.minDimension * .39f * pulse, center = center,
                style = Stroke(width = 3.dp.toPx()))
            drawCircle(color = Violet.copy(alpha = if (listening) .22f else .08f),
                radius = size.minDimension * .46f, center = center,
                style = Stroke(width = 1.3.dp.toPx()))
        }
        Column(
            modifier = Modifier.fillMaxSize().verticalScroll(rememberScrollState())
                .padding(horizontal = 29.dp, vertical = 15.dp),
            horizontalAlignment = Alignment.CenterHorizontally,
            verticalArrangement = Arrangement.Center
        ) {
            Word("vibra", 19, Ice, FontWeight.Bold)
            Spacer(Modifier.height(3.dp))
            Word(if (state.connected) "● Celular conectado" else "○ Sin conexión", 10,
                if (state.connected) Cyan else Muted)
            Spacer(Modifier.height(8.dp))
            Box(Modifier.size(82.dp).clip(CircleShape)
                .background(Brush.linearGradient(listOf(Cyan, Color(0xFF136CFB), Violet)))
                .clickable(onClick = onRecognize), contentAlignment = Alignment.Center) {
                Image(painterResource(R.drawable.vibra_mark), "Reconocer canción",
                    Modifier.size(72.dp))
            }
            Spacer(Modifier.height(8.dp))
            val headline = when (state.stage) {
                "requesting" -> "Conectando..."
                "waiting" -> "En tu celular"
                "recording" -> "Escuchando..."
                "identifying" -> "Identificando..."
                "found" -> "¡La encontré!"
                "no_match" -> "Sin coincidencia"
                "error" -> "Revisa conexión"
                else -> "Reconocer música"
            }
            Word(headline, 14, Ice, FontWeight.Bold)
            if (state.stage == "found" && state.title.isNotBlank()) {
                Spacer(Modifier.height(3.dp))
                Word(state.title, 12, Cyan, FontWeight.SemiBold)
                Word(state.artist, 10, Muted)
            } else {
                Spacer(Modifier.height(4.dp))
                Word(state.detail, 10, Muted)
            }
            if (state.stage == "found" || state.stage == "error" || state.stage == "no_match") {
                Spacer(Modifier.height(5.dp))
                Word("Toca el icono para buscar otra", 9, Muted)
            }
        }
    }
}

@Composable
private fun Word(text: String, size: Int, color: Color, weight: FontWeight = FontWeight.Normal) {
    BasicText(text, modifier = Modifier.fillMaxWidth(), maxLines = 2, overflow = TextOverflow.Ellipsis,
        style = TextStyle(color = color, fontSize = size.sp, fontWeight = weight,
            textAlign = TextAlign.Center, lineHeight = (size + 3).sp))
}
